172: // #873 / #654: on some Windows setups the first hipMemGetInfo of a process fails with hipErrorInvalidValue because the ROCm
173: // runtime has not finished initialising against the Adrenalin WDDM driver, and the first kernel launch after that
174: // crashes (0xC0000005 in amdhip64_7.dll).  A hipFree(0) before the first query initialises the context, and one
175: // failed query gets one retry after 250 ms.  Both happen once per process, before the DXGI budget below; any other
176: // error, and a failure of the retry, are returned as they are (a real failure is never retried).
177: static hipError_t first_mem_get_info(size_t* free_bytes, size_t* total_bytes) {
178:     static std::atomic<bool> warmed{false}, retried{false};
179:     if (!warmed.exchange(true)) (void) hipFree(nullptr);
180:     hipError_t e = hipMemGetInfo(free_bytes, total_bytes);
181:     if (e == hipErrorInvalidValue && !retried.exchange(true)) {
182:         (void) hipGetLastError();
183:         std::fprintf(stderr, "strata: hipMemGetInfo failed (%s); retrying once after 250 ms\n", hipGetErrorString(e));
184:         Sleep(250);
185:         e = hipMemGetInfo(free_bytes, total_bytes);
186:         std::fprintf(stderr, "strata: hipMemGetInfo retry: %s\n", e == hipSuccess ? "ok" : hipGetErrorString(e));
187:     }
188:     return e;
189: }
